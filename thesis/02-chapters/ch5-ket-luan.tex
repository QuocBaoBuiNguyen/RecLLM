\chapter{Kết luận và hướng nghiên cứu}
\label{Chapter5}

\section{Tóm tắt đóng góp}
\label{sec:tom-tat-dong-gop}

Luận văn đặt ra câu hỏi: liệu mô-đun căn chỉnh Q-Former -- với tập truy vấn học được -- có thể
căn chỉnh tín hiệu cộng tác vào không gian ngữ nghĩa của LLM hiệu quả
hơn mô-đun chiếu MLP của CoLLM hay không? Để trả lời, luận văn thiết kế, hiện thực và đánh
giá mô hình CoQLLM, đóng góp theo hai hướng đã phát biểu ở Mục~\ref{sec:dong-gop}:

\textbf{Đóng góp 1: Mô-đun căn chỉnh Q-Former được thích nghi và tăng cường cho bài toán gợi ý.}
CoQLLM thay thế mô-đun chiếu MLP pointwise của CoLLM bằng kiến trúc Q-Former, trong đó embedding
cộng tác của sản phẩm mục tiêu và của các sản phẩm trong lịch sử thay thế đặc trưng ảnh làm đầu
vào cross-attention, tín hiệu người dùng được đưa vào qua điều kiện hóa truy vấn, và toàn bộ
được chuyển thành soft token cho LLM. Trên nền đó, luận văn bổ sung kỹ thuật huấn luyện điều
kiện hóa truy vấn theo người dùng, cùng với biểu diễn đa token như một hệ quả của thiết kế
nhiều truy vấn. Kết quả chi tiết được trình bày ở Chương~\ref{Chapter4}.

\textbf{Đóng góp 2: Đánh giá có hệ thống theo độ đo AUC/uAUC trên hai tập dữ liệu.}
Luận văn đánh giá mô-đun căn chỉnh Q-Former trên bài toán CTR với AUC và uAUC -- cùng quy trình với
CoLLM -- trên MovieLens-1M và Amazon-Book, cho phép so sánh trực tiếp với CoLLM, BinLLM và
CoRA. Phân tích ablation tách biệt đóng góp của từng kỹ thuật, và phân rã warm/cold làm rõ
trần uAUC nằm ở chất lượng embedding MF.

\section{Kết quả chính}
\label{sec:ket-qua-dat-duoc}

Từ kết quả ở Bảng~\ref{tab:ket-qua-chinh}, Bảng~\ref{tab:ket-qua-book} và
Bảng~\ref{tab:ablation}, có thể rút ra các kết luận sau:

\begin{itemize}
    \item \textbf{MovieLens-1M -- vượt SOTA trên cả hai độ đo.} Cấu hình đề xuất
    (+ điều kiện hóa) đạt AUC $0{,}7475$ và uAUC $0{,}6968$, vượt BinLLM
    ($0{,}7425$/$0{,}6956$) trên cả AUC lẫn uAUC và vượt rõ CoLLM. Đây là điểm đáng chú ý
    vì nhiều phương pháp cùng hướng chỉ thắng một trong hai độ đo.

    \item \textbf{Amazon-Book.} CoQLLM đạt $0{,}8134$/$0{,}6225$ (số re-run), cao hơn CoLLM-MF và
    BinLLM (theo lần chạy lại của CoRA) trên cả hai độ đo và sát CoRA-MF ($0{,}8179$/$0{,}6262$).

    \item \textbf{Chèn soft token cộng tác cải thiện cả AUC và uAUC.} Thí nghiệm loại bỏ thành
    phần cho thấy việc đưa tín hiệu cộng tác vào LLM qua soft token làm tăng cả hai độ đo trên hai
    tập dữ liệu, rõ nhất trên Amazon-Book (Bảng~\ref{tab:ablation}).

    \item \textbf{uAUC và nhóm cold.} Phân rã warm/cold trên MovieLens-1M cho thấy uAUC nhóm
    warm ($0{,}7254$) cao hơn nhóm cold ($0{,}5969$); quan sát này gợi ý hướng cải thiện uAUC
    tiếp theo là nâng chất lượng embedding MF cho người dùng/sản phẩm nguội.
\end{itemize}

\section{Hạn chế nghiên cứu}
\label{sec:han-che-ch5}

Luận văn còn hai hạn chế chính:

\begin{enumerate}
    \item \textbf{Chi phí huấn luyện cao hơn:} Mô-đun căn chỉnh Q-Former có kiến trúc nhiều lớp
    và số tham số lớn hơn lớp ánh xạ MLP của CoLLM, nên chi phí huấn luyện cao hơn (dù backbone
    LLM vẫn được giữ đóng băng và không sinh trọng số theo từng mẫu như CoRA).

    \item \textbf{Hiệu suất chưa đạt mức tốt nhất trên mọi thiết lập:} Trên Amazon-Book, CoQLLM
    cạnh tranh nhưng chưa vượt CoRA-MF ở cả hai độ đo. Khoảng cách này phản ánh một số hạn chế
    còn lại trong thiết kế mô hình, chẳng hạn mỗi sản phẩm chỉ được mã hóa từ một véc-tơ cộng tác
    và chất lượng embedding MF cho nhóm người dùng/sản phẩm nguội còn thấp.
\end{enumerate}

\section{Hướng nghiên cứu tương lai}
\label{sec:huong-phat-trien}

Từ các quan sát và hạn chế trên, luận văn đề xuất một số hướng nghiên cứu tiếp theo.

\textbf{Ablation và phân tích sâu hơn.}
\begin{itemize}
    \item Bổ sung hàng ablation attention-MLP (bộ chiếu có self-attention nhưng không có truy
    vấn học được) để cô lập đóng góp của cấu trúc Q-Former khỏi phần phi tuyến và attention đơn
    thuần.
    \item Phân tích attention map của Q-Former để hiểu các truy vấn học cách trích xuất thông tin
    gì từ tín hiệu cộng tác.
\end{itemize}

\textbf{Cải thiện uAUC.}
\begin{itemize}
    \item Tiền huấn luyện MF trên dữ liệu dày đặc hơn, đặc biệt cho nhóm người dùng/sản phẩm
    nguội, nhằm nâng chất lượng biểu diễn cộng tác đầu vào.
    \item Khảo sát các mục tiêu huấn luyện hướng tới thứ tự xếp hạng trong từng người dùng (chẳng
    hạn tổn thất bảo toàn thứ hạng theo cặp).
\end{itemize}

\textbf{Mở rộng kiến trúc.}
\begin{itemize}
    \item Sử dụng các mô hình cộng tác cung cấp \textit{nhiều} véc-tơ đặc trưng cho mỗi sản phẩm
    thay vì một véc-tơ duy nhất (chẳng hạn biểu diễn theo nhiều khía cạnh hoặc theo chuỗi tương
    tác), để cross-attention của Q-Former có thể chọn lọc trên nhiều thành phần thay vì chỉ đọc
    một véc-tơ.
    \item Áp dụng thêm các kỹ thuật huấn luyện và kiến trúc cho Q-Former ngoài điều kiện hóa truy
    vấn theo người dùng, chẳng hạn tổn thất bảo toàn thứ hạng trên soft token, truy vấn nhận biết
    sản phẩm ứng viên, hoặc chèn token biểu diễn người dùng.
    \item Khảo sát các backbone LLM khác (chẳng hạn Llama-3, Mistral) để kiểm chứng tính tổng
    quát của mô-đun căn chỉnh Q-Former.
\end{itemize}
